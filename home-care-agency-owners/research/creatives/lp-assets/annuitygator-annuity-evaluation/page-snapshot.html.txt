        function formatPhoneNumber(value) {
            const digits = normalizeUsPhoneDigits(value);

            if (!digits.length) return '';

            let country = '';
            let phone = digits;

            // Detect US country code
            if (digits.length === 11 && digits.startsWith('1')) {
                country = '1 ';
                phone = digits.slice(1);
            }

            const area = phone.slice(0, 3);
            const prefix = phone.slice(3, 6);
            const line = phone.slice(6, 10);

            let formatted = '';

            if (phone.length <= 3) {
                formatted = `(${area}`;
            } else if (phone.length <= 6) {
                formatted = `(${area}) ${prefix}`;
            } else {
                formatted = `(${area}) ${prefix}-${line}`;
            }

            return country + formatted;
        }

        async function verifyEmailField(validator, options = {}) {
            const { force = false } = options;
            const email = document.querySelector("[name='email']").value.trim();

            if (!email) {
                verificationState.email = { status: 'idle', value: '', message: '', requestPromise: null };
                return false;
            }

            if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
                verificationState.email = { status: 'invalid', value: email, message: 'Enter a valid email address', requestPromise: null };
                return false;
            }

            let requestPromise = null;

            if (!force && verificationState.email.value === email) {
                if (verificationState.email.status === 'valid') {
                    return true;
                }
                if (verificationState.email.status === 'invalid') {
                    return false;
                }
                if (verificationState.email.status === 'pending' && verificationState.email.requestPromise) {
                    requestPromise = verificationState.email.requestPromise;
                }
            }

            if (!requestPromise) {
                requestPromise = verifyField('email', email);
                verificationState.email = { status: 'pending', value: email, message: '', requestPromise };
            }

            try {
                const isValid = await requestPromise;

                if (verificationState.email.value !== email) {
                    return false;
                }

                if (isValid) {
                    verificationState.email = { status: 'valid', value: email, message: '', requestPromise: null };
                    clearValidationError("[name='email']");
                    return true;
                }

                verificationState.email = { status: 'invalid', value: email, message: 'Email is invalid', requestPromise: null };
                validator.showErrors({
                    "[name='email']": 'Email is invalid'
                });
                return false;
            } catch (error) {
                if (error.name === 'AbortError') {
                    verificationState.email = { status: 'idle', value: '', message: '', requestPromise: null };
                    return true;
                }
                console.error('Email verification failed:', error);
                verificationState.email = { status: 'error', value: email, message: 'Could not validate email right now. Please try again.', requestPromise: null };
                validator.showErrors({
                    "[name='email']": verificationState.email.message
                });
                return false;
            }
        }

            const phoneRaw = document.querySelector("[name='phone']").value;
            const phoneDigits = phoneRaw.replace(/\D/g, '');

            // Normalizar: quitar country code si tiene 11 dígitos y empieza con 1
            const normalizedDigits = phoneDigits.length === 11 && phoneDigits.startsWith('1')
                ? phoneDigits.slice(1)
                : phoneDigits;

            if (!normalizedDigits) {
                verificationState.phone = { status: 'idle', value: '', message: '', requestPromise: null };
                return false;
            }

            if (normalizedDigits.length !== 10) {
                verificationState.phone = { status: 'invalid', value: normalizedDigits, message: 'Phone number must be in the format (123) 456-7890', requestPromise: null };
                return false;
            }

            let requestPromise = null;

            if (!force && verificationState.phone.value === normalizedDigits) {
                if (verificationState.phone.status === 'valid') {
                    return true;
                }
                if (verificationState.phone.status === 'invalid') {
                    return false;
                }
                if (verificationState.phone.status === 'pending' && verificationState.phone.requestPromise) {
                    requestPromise = verificationState.phone.requestPromise;
                }
            }

            if (!requestPromise) {
                requestPromise = verifyField('phone', normalizedDigits);
                verificationState.phone = { status: 'pending', value: normalizedDigits, message: '', requestPromise };
            }

            try {
                const isValid = await requestPromise;

                if (verificationState.phone.value !== normalizedDigits) {
                    return false;
                }

                if (isValid) {
                    verificationState.phone = { status: 'valid', value: normalizedDigits, message: '', requestPromise: null };
                    clearValidationError("[name='phone']");
                    return true;
                }

                verificationState.phone = { status: 'invalid', value: normalizedDigits, message: 'Phone number is invalid', requestPromise: null };
                validator.showErrors({
                    "[name='phone']": 'Phone number is invalid'
                });
                return false;
            } catch (error) {
                if (error.name === 'AbortError') {
                    verificationState.phone = { status: 'idle', value: '', message: '', requestPromise: null };
                    return true;
                }
                console.error('Phone verification failed:', error);
                verificationState.phone = { status: 'error', value: normalizedDigits, message: 'Could not validate phone right now. Please try again.', requestPromise: null };
                validator.showErrors({
                    "[name='phone']": verificationState.phone.message
                });
                return false;
            }

            const area = phone.slice(0, 3);
            const prefix = phone.slice(3, 6);
            const line = phone.slice(6, 10);

            let formatted = '';

            if (phone.length <= 3) {
                formatted = `(${area}`;
            } else if (phone.length <= 6) {
                formatted = `(${area}) ${prefix}`;
            } else {
                formatted = `(${area}) ${prefix}-${line}`;
            }

            return country + formatted;
        }

        function setWithExpiry(key, value, ttlMinutes) {
            const now = new Date();

            const item = {
                value: value,
                expiry: now.getTime() + ttlMinutes * 60 * 1000
            };

            localStorage.setItem(key, JSON.stringify(item));
        }

        function getWithExpiry(key) {
            const itemStr = localStorage.getItem(key);

            if (!itemStr) return null;

            const item = JSON.parse(itemStr);
            const now = new Date();

            if (now.getTime() > item.expiry) {
                localStorage.removeItem(key);
                return null;
            }

            return item.value;
        }

        async function sendToEndpoint(url, data, headers) {
            const controller = new AbortController();
            const timeoutId = setTimeout(() => controller.abort(), 45000);
            try {
                const response = await fetch(url, {
                    method: 'POST',
                    signal: controller.signal,
                    headers: {
                        'Content-Type': 'application/json',
                        'Accept': 'application/json',
                        ...headers
                    },
                    body: JSON.stringify(data)
                });
                clearTimeout(timeoutId);

                if (!response.ok) {
                    throw new Error(`HTTP error! Status: ${response.status}`);
                }

                const responseData = await response.json();
                return { success: true, responseData };

            } catch (error) {
                clearTimeout(timeoutId);
                console.log('Error al realizar la solicitud:', error, url, data, headers);
                throw error;
            }
        }
